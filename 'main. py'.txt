    <div id="channel-sec" class="card">
      <h2>Create IndiaTube Channel</h2>
      <input type="text" id="channel-input" placeholder="Channel Name">
      <button class="btn-main" onclick="createChannel()">Save Channel</button>

      <input type="text" id="vid-title" placeholder="Video Title">
      <input type="file" id="vid-file" accept="video/*">
      <button class="btn-main" onclick="uploadVideo()">Publish</button>

  <script>
    let currentUser = localStorage.getItem('it_user') || null;
    let userChannel = localStorage.getItem('it_channel') || null;
    let videos = JSON.parse(localStorage.getItem('it_videos')) || [];

    function updateNav() {
      const authBtn = document.getElementById('auth-btn');
      if (currentUser) {
        authBtn.innerText = userChannel ? userChannel : currentUser;
      } else {
        authBtn.innerText = "Login";
      }
    }

    function showSection(id) {
      document.querySelectorAll('.card').forEach(el => el.classList.remove('active'));
      document.getElementById(id).classList.add('active');
    }

    function handleAuth() {
      const name = document.getElementById('username-input').value.trim();
      if (!name) return alert('Kripya naam enter karein!');
      
      currentUser = name;
      localStorage.setItem('it_user', name);
      updateNav();

      if (!userChannel) {
        showSection('channel-sec');
      } else {
        showSection('feed-sec');
      }
    }

    function createChannel() {
      const chName = document.getElementById('channel-input').value.trim();
      if (!chName) return alert('Channel ka naam likhein!');
      
      userChannel = chName;
      localStorage.setItem('it_channel', chName);
      updateNav();
      alert('Channel create ho gaya: ' + chName);
      showSection('feed-sec');
    }

    function uploadVideo() {
      if (!currentUser) {
        alert('Upload karne se pehle login karein!');
        showSection('auth-sec');
        return;
      }
      if (!userChannel) {
        alert('Pehle apna channel banayein!');
        showSection('channel-sec');
        return;
      }

      const title = document.getElementById('vid-title').value.trim();
      const fileInput = document.getElementById('vid-file');
      const file = fileInput.files[0];

      if (!title || !file) {
        return alert('Title aur Video file dono select karein!');
      }

      const videoUrl = URL.createObjectURL(file);
      const newVideo = {
        title: title,
        channel: userChannel,
        url: videoUrl
      };

      videos.unshift(newVideo);
      document.getElementById('vid-title').value = '';
      fileInput.value = '';

      renderVideos();
      showSection('feed-sec');
    }

    function renderVideos() {
      const list = document.getElementById('video-list');
      if (videos.length === 0) {
        list.innerHTML = '<p style="color: #888;">No videos uploaded yet.</p>';
        return;
      }
      
      list.innerHTML = '';
      videos.forEach(v => {
        const item = document.createElement('div');
        item.className = 'video-card';
        item.innerHTML = `
          <video src="${v.url}" controls></video>
          <div class="video-info">
            <div class="video-title">${v.title}</div>
            <div class="video-channel">${v.channel}</div>
          </div>
        `;
        list.appendChild(item);
      });
    }

    updateNav();
    renderVideos();
  </script>
